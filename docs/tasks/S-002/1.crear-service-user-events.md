# Task S-002.1: Crear Service para User Events

## Status

**Completed**

## Description

Agregar el nuevo endpoint y método en el servicio API del frontend para consultar los eventos donde un usuario participa.

## Acceptance Criteria

1. Existe el endpoint `USER_EVENTS` en `api.ts`
2. Existe el método `getUserEvents()` en `services/api.ts`
3. El método devuelve eventos en formato `Event[]` del frontend
4. El método maneja errores correctamente

## Technical Details

**Archivos a Modificar:**

### `src/config/api.ts`

```typescript
export const API_CONFIG = {
  BASE_URL: process.env.REACT_APP_API_URL || 'http://localhost:8080',
  ENDPOINTS: {
    // ... existing endpoints ...
    USER_EVENTS: (userId: string) => `/api/v1/users/${userId}/events`,
  }
};
```

### `src/services/api.ts`

```typescript
export const UserService = {
  async getUserEvents(userId: string): Promise<Event[]> {
    try {
      const response = await apiRequest(API_CONFIG.ENDPOINTS.USER_EVENTS(userId));
      
      if (!response.data || !Array.isArray(response.data)) {
        return [];
      }

      return response.data.map(event => ({
        id: event.id,
        title: event.name || event.title,
        description: event.description,
        stage: event.stage as Event['stage'],
        date: event.start_date || event.date,
        organizer: event.organizer,
        shareable_link: event.shareable_link,
        status: event.status || 'active',
        participant_ids: event.participant_ids,
        attachmentCount: event.attachment_count || 0,
        creator_id: event.author_id || event.creator_id,
        created_at: event.created_at,
        updated_at: event.updated_at,
      }));
    } catch (error) {
      console.error('Error fetching user events:', error);
      throw error;
    }
  }
};
```
